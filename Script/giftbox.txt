procedure OnDblClick(aStr : String);
var
   Str, Name, Item : String;
   Num : Integer;
begin
   // 原则上不需要判断，但防其它物品调用脚本BUG
   Str := callfunc ('getsenderitemexistence 礼盒:1');
   if Str = 'false' then begin
      exit;
   end;
   Str := callfunc ('checkenoughspace');
   if Str = 'false' then begin
      print('sendsenderchatmessage 物品栏没有空间放礼物啦 1');
      exit;
   end;
   // 打开回收
   print ('getsenderitem 礼盒:1');
   // 获取奖励
   Item := callfunc ('getrandomitem 0');
   Num := Random (15);
   if Num > 2 then Item := callfunc ('getrandomitem 2');
   if Num > 6 then Item := callfunc ('getrandomitem 3');
   if Num > 12 then Item := callfunc ('getrandomitem 4');
   Str := 'putsendermagicitem '+ Item;
   Str := Str + ' @礼盒 2';
   print (Str);
   // 全服嘚瑟
   Name := callfunc ('getsendername');
   Str := 'sendsendertopmsg 玩家_' + Name;
   Str := Str + '_打开礼盒,获得_';
   Str := Str + Item;
   print (Str);
end;

end.
