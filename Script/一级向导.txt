procedure OnApproach (aStr : String);
var
   Str : String;
   Race : Integer;
begin
   Str := callfunc ('getsenderrace');
   Race := StrToInt (Str);
   if Race = 1 then begin
      print ('say 新玩家可以找我了解游戏玩法，完成任务有奖励哦');
      // Str := 'showwindow .\help\start.txt';
      // print (str);
      exit;
   end;
end;

procedure OnAway (aStr : String);
var
   Str, PowerLevel, Name : String;
   Race, Lv : Integer;
begin
   Str := callfunc ('getsenderrace');
   Race := StrToInt (Str);
   if Race = 1 then begin
      Name := callfunc ('getsendername');
      if Name = '秋林拾叶' then begin
         print ('say ^_^');
         print ('senderrefill');
         exit;
      end;
      // 境界
      PowerLevel := callfunc ('getsendercurpowerlevel');
      Lv := StrToInt (PowerLevel);
      if Lv >= 4 then begin
         Str := 'say ' + Name;
         Str := Str + '，你的境界太高了，我还是送您出村吧。';
         print (Str);
         Str := 'movespace ' + Name;
         Str := Str + ' user 0 180 240 300';
         print (Str);
         exit;
      end;
      // 年龄
      Str := callfunc ('getsenderage');
      Lv := StrToInt(Str);
      if Lv > 4000 then begin
         Str := 'say ' + Name;
         Str := Str + '，你这么大了怎么还在这里，我送您出村吧。';
         print (Str);
         Str := 'movespace ' + Name;
         Str := Str + ' user 0 180 240 300';
         print (Str);
         exit;
      end;
   end;
end;

   end;

         exit;

            exit;

         Str := callfunc ('checkenoughspace 1');

            print ('say 物品栏已满，无法领取任务奖励');
            exit;
         end;
         print ('putsendermagicitem 初级宝箱:1 @一级向导 4');
         print ('changesendercompletequest 1');
         print ('changesendercurrentquest 2');

         exit;

            exit;

      // 任务3
