// 动态物品测试脚本，GM控制台，不得用于玩家场景
procedure OnHear (aStr : String);
var
   Str, Eval, Cmd, Arg, Arg1, Arg2, Arg3, Arg4, Arg5, Arg6, Arg7, Res : String;
   Race : Integer;
begin
   Str := callfunc ('getsenderrace');
   Race := StrToInt (Str);
   if Race = 1 then begin
      // GM测试指令：call getsendersex 或 print say hi，以call和print开头
      // 使用 GetToken 函数将 aStr 以 空格'_' 分隔为为子字符串，aToken为结果，返回值为剩余部分
      Cmd := GetToken (aStr, Eval, '_');
      // 函数输入参数如果有空格，必须重新组合，否则被当成一个字符串处理了
      Arg := GetToken (Cmd, Cmd, '_');
      if Eval = 'call' then begin
         if Arg <> '' then begin
            // callfunc最多只有三个参数(checkusemagicbygrade)
            Arg := GetToken (Arg, Arg1, '_');
            Arg := GetToken (Arg, Arg2, '_');
            Arg := GetToken (Arg, Arg3, '_');
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg1;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg2;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg3;
         end;
         // 输出函数结果
         Res := callfunc (Cmd); Res := Res + ' 1';
         Str := 'sendsenderchatmessage ' + Res;
         print(Str);
         exit;
      end;
      if Eval = 'print' then begin
         if Arg <> '' then begin
            // print最多存在8个参数(mapaddobjbytick和movespacebyname)
            // 游戏脚本不支持数组，无法用for循环简化代码
            Arg := GetToken (Arg, Arg1, '_');
            Arg := GetToken (Arg, Arg2, '_');
            Arg := GetToken (Arg, Arg3, '_');
            Arg := GetToken (Arg, Arg4, '_');
            Arg := GetToken (Arg, Arg5, '_');
            Arg := GetToken (Arg, Arg6, '_');
            Arg := GetToken (Arg, Arg7, '_');
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg1;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg2;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg3;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg4;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg5;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg6;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg7;
            Cmd := Cmd + ' ';
            Cmd := Cmd + Arg;
         end;
         print(Cmd);
         exit;
      end;
      exit;
   end;
end;
